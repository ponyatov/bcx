/// @brief opcode
enum class Op : byte {
    nop = 0x00,   ///< @ref nop
    halt = 0xFF,  ///< @ref halt
    jmp = 0x01,   ///< @ref jmp
    jz = 0x02,    ///< @ref jz
    call = 0x03,  ///< @ref call

};
